              c.ci_status = 'PENDING', c.timestamp = $timestamp

MATCH (parent:Commit {hash: p_hash})
MERGE (c)-[:PARENT]->(parent);
// C. READ 1: BOUNDED DAG LINEAGE FOR NEXT.JS GRAPH CANVAS
